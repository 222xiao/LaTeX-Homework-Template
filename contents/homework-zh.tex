\problem{公式推导}
\label{sec:mean}
给定 $n$ 个观测值 $x_1,\ldots,x_n$，求使平方误差最小的常数 $a$。
\begin{solution}
目标函数与一阶导数分别为
\begin{align}
  L(a) &= \sum_{i=1}^{n}(x_i-a)^2, \\
  \frac{\mathrm{d}L}{\mathrm{d}a} &= -2\sum_{i=1}^{n}(x_i-a)=0.
\end{align}
因此最优解是样本均值：
\begin{equation}
  a^*=\bar{x}=\frac{1}{n}\sum_{i=1}^{n}x_i.
  \label{eq:mean}
\end{equation}
由于 $L''(a)=2n>0$（$n\geq1$），该解为唯一最小值点。
\end{solution}
